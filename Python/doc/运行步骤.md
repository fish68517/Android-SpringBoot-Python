### 最终版 `face_api.py` 脚本

这个版本有以下几个关键改进：

1. **模型预加载**：在服务启动时就将 `ArcFace` 模型加载到内存中。这可以避免第一个用户访问时需要等待漫长的模型加载时间，极大提升响应速度。
2. **更严谨的登录逻辑**：在 `/login` 接口中，我们不仅找到最匹配的人，还会检查其 `distance` (差异度) 是否小于 `threshold` (阈值)。这可以有效防止把一个陌生人错认为数据库里的某个用户。
3. **统一的目录管理**：将文件夹路径统一定义在配置区，方便修改。
4. **清晰的JSON响应**：为所有情况（成功、失败、错误）设计了清晰一致的 JSON 返回格式，方便你的 Android App 进行解析。

将以下代码保存为 `face_api.py` (或者你喜欢的任何名字，比如 `app.py`)：

```python
import os
import uuid
import traceback
from flask import Flask, request, jsonify
from deepface import DeepFace

# --- 1. 全局配置 ---
app = Flask(__name__)

# 用于存储注册人脸的 "数据库" 文件夹
DB_PATH = "face_database" 
# 用于临时存放上传文件的文件夹
UPLOAD_FOLDER = "temp_uploads" 
os.makedirs(DB_PATH, exist_ok=True)
os.makedirs(UPLOAD_FOLDER, exist_ok=True)

# DeepFace 模型配置
MODEL_NAME = "ArcFace"
DETECTOR_BACKEND = "mtcnn"
DISTANCE_METRIC = "cosine" # 距离度量方式

# 禁用 deepface 的详细日志，保持控制台干净
os.environ['DEEPFACE_LOG_LEVEL'] = 'ERROR'

# --- 2. 预加载模型 ---
# 在服务启动时，预先构建并加载模型到内存。
# 这样可以确保API接口的响应速度，避免每次请求都重新加载。
print("正在加载人脸识别模型...")
try:
    DeepFace.build_model(MODEL_NAME)
    print(f"模型 '{MODEL_NAME}' 加载成功，服务已准备就绪！")
except Exception as e:
    print(f"模型加载失败，错误: {e}")
    # 如果模型加载失败，服务无法正常工作，可以选择退出
    # exit()


# --- 3. API 接口定义 ---

@app.route("/register", methods=["POST"])
def register():
    """
    人脸注册接口
    接收表单数据: 'user_id' 和 'image' 文件
    """
    # 检查请求中是否包含必要的数据
    if 'image' not in request.files:
        return jsonify({"status": "error", "message": "请求中未找到 'image' 文件"}), 400
    if 'user_id' not in request.form:
        return jsonify({"status": "error", "message": "请求中未找到 'user_id' 参数"}), 400

    image_file = request.files['image']
    user_id = request.form['user_id']
  
    # 检查 user_id 是否为空
    if not user_id.strip():
        return jsonify({"status": "error", "message": "'user_id' 不能为空"}), 400

    # 使用 user_id 作为主文件名，确保唯一性。注意：这会覆盖同名用户的注册照片。
    # 实际项目中可能需要更复杂的逻辑，但对于毕设这已足够。
    image_filename = f"{user_id}.jpg"
    image_path = os.path.join(DB_PATH, image_filename)
    image_file.save(image_path)

    try:
        # 验证保存的图片中是否能检测到人脸，这是保证数据质量的关键步骤
        DeepFace.represent(
            img_path=image_path,
            model_name=MODEL_NAME,
            detector_backend=DETECTOR_BACKEND,
            enforce_detection=True # 强制要求必须检测到人脸
        )
      
        print(f"SUCCESS: 用户 '{user_id}' 注册成功, 图片保存至 '{image_path}'")
        return jsonify({"status": "success", "message": f"用户 '{user_id}' 注册成功"})

    except ValueError:
        # 如果 represent 找不到人脸，会抛出 ValueError
        os.remove(image_path) # 从数据库中删除这张无效图片
        print(f"FAIL: 用户 '{user_id}' 注册失败，图片中未检测到人脸。")
        return jsonify({"status": "error", "message": "注册失败: 提供的图片中无法检测到清晰的人脸"}), 400
    except Exception as e:
        if os.path.exists(image_path):
            os.remove(image_path)
        traceback.print_exc() # 在服务器后台打印完整的错误堆栈，方便调试
        return jsonify({"status": "error", "message": f"服务器内部错误: {str(e)}"}), 500


@app.route("/login", methods=["POST"])
def login():
    """
    人脸登录接口
    接收 'image' 文件
    """
    if 'image' not in request.files:
        return jsonify({"status": "error", "message": "请求中未找到 'image' 文件"}), 400

    image_file = request.files['image']
  
    # 使用UUID确保临时文件名唯一，避免并发请求时文件被覆盖
    temp_image_path = os.path.join(UPLOAD_FOLDER, f"{uuid.uuid4()}.jpg")
    image_file.save(temp_image_path)

    try:
        # 在人脸数据库(DB_PATH)中进行1:N搜索
        # DeepFace.find 会返回一个包含DataFrame的列表
        dfs = DeepFace.find(
            img_path=temp_image_path,
            db_path=DB_PATH,
            model_name=MODEL_NAME,
            detector_backend=DETECTOR_BACKEND,
            distance_metric=DISTANCE_METRIC,
            enforce_detection=True, # 同样强制要求登录图片必须有人脸
            silent=True
        )

        # 检查是否找到了任何匹配项
        if len(dfs) > 0 and not dfs[0].empty:
            # dfs[0] 是一个 pandas DataFrame，结果已按 distance 从小到大排好序
            best_match = dfs[0].iloc[0].to_dict()
          
            # 从文件路径中提取 user_id (即文件名，不含扩展名)
            identity_path = best_match.get('identity', '')
            user_id = os.path.splitext(os.path.basename(identity_path))[0]
          
            # 【核心验证逻辑】
            # 获取这次匹配的差异度(distance)和模型推荐的阈值(threshold)
            distance = best_match.get(f'{MODEL_NAME}_{DISTANCE_METRIC}', 1.0)
            threshold = DeepFace.get_threshold(MODEL_NAME, DISTANCE_METRIC)
          
            if distance <= threshold:
                # 只有当差异度小于等于阈值时，才认为是同一个人
                print(f"SUCCESS: 登录成功! 识别为用户 '{user_id}', Distance: {distance:.4f}, Threshold: {threshold}")
                return jsonify({
                    "status": "success", 
                    "user_id": user_id,
                    "distance": distance
                })
            else:
                # 找到了相似的人，但相似度不够高，不足以确认为同一个人
                print(f"FAIL: 登录失败. 最佳匹配 '{user_id}' 的相似度不足. Distance: {distance:.4f}, Threshold: {threshold}")
                return jsonify({"status": "failure", "message": "人脸不匹配，请重试"})
        else:
            # 数据库为空，或没有任何一张脸的相似度高到可以通过find函数的内部筛选
            print("FAIL: 登录失败. 数据库中未找到任何匹配的人脸。")
            return jsonify({"status": "failure", "message": "用户未注册或人脸不匹配"})

    except ValueError:
        # 上传的登录图片中未检测到人脸
        print("FAIL: 登录失败，上传的图片中未检测到人脸。")
        return jsonify({"status": "error", "message": "登录失败: 无法检测到清晰的人脸"}), 400
    except Exception as e:
        traceback.print_exc()
        return jsonify({"status": "error", "message": f"服务器内部错误: {str(e)}"}), 500
    finally:
        # 无论成功失败，最后都要清理临时文件
        if os.path.exists(temp_image_path):
            os.remove(temp_image_path)

# --- 4. 启动服务 ---
if __name__ == '__main__':
    # host='0.0.0.0' 让服务可以被局域网内的其他设备（比如你的手机）访问
    # debug=True 可以在开发时提供有用的调试信息，但在最终部署时应设为 False
    app.run(host='0.0.0.0', port=5000, debug=True)
```

### 如何运行和使用

1. **安装依赖库**：
   在你的 Anaconda 环境 (`face_rec_env`) 中，确保已安装 `Flask` 和 `deepface`。

   ```bash
   conda activate face_rec_env
   pip install Flask deepface
   ```
2. **启动服务**：
   将上面的代码保存为 `face_api.py`，然后在终端中运行它。

   ```bash
   python face_api.py
   ```

   你会看到模型加载的提示，最后服务会运行在 `http://0.0.0.0:5000/` 上。
3. **Android端如何访问**：

   * 首先，确保你的电脑和Android手机连接在 **同一个Wi-Fi网络** 下。
   * 在电脑的命令行/终端中，使用 `ipconfig` (Windows) 或 `ifconfig` (Mac/Linux) 命令查找到你电脑的 **局域网IP地址** (通常是 `192.168.x.x` 格式)。
   * 在你的Android App中，所有API请求的地址就应该是 `http://<你电脑的IP地址>:5000`。
     * 注册接口地址: `http://192.168.1.10:5000/register`
     * 登录接口地址: `http://192.168.1.10:5000/login`

现在，你的Python后端已经是一个功能完整、稳定可靠的人脸识别微服务了。你可以专注于Android端的开发，通过HTTP请求与它进行交互即可。
